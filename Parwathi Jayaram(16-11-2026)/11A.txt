So for this question I iteratively checked if the values are increasing.
Otherwise I calculated how many times d needs to be added.
#include <bits/stdc++.h>
using namespace std;
int main() {
    int n,d;
    cin>>n>>d;
    vector<int> b;
    for(int i=0;i<n;i++){
        int a;
        cin>>a;
        b.push_back(a);
    }
    int prev=b[0];
    int op=0;
    for(int i=1;i<n;i++){        
        if(b[i]<=prev){
        int x=(prev-b[i])/d+1;
        b[i]+=x*d;
        op+=x;
    }
        prev=b[i];
    }
    cout<<op<<endl;
    return 0;
}
