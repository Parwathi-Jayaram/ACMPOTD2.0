So for this question I first counted the occuerence of each character in the string.
Then I made a string where all these characters will occur count/n times.
If count is not divisible by n that means that this string can't be written as a k-string.
#include <bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin>>n;
    string s;
    cin>>s;
    int min=0;
    map<char,int>mp;
    for(char c:s){
        mp[c]++;
    }
    string op="";
    for(auto &x:mp){
        char a=x.first;
        int b=x.second;
        if(b%n!=0){
            cout<<-1<<endl;
            return 0;
        }
        op+=string(b/n,a);
    }
    string ans="";
    for(int i=0;i<n;i++){
        ans+=op;
    }
    cout<<ans<<endl;
    return 0;
}
