So for this question I actually created an empty string first. 
Then I iterated through the input string and if the character is '-' then I checked the next character to see if its 2 or 1 otherwise it is 0.
#include <bits/stdc++.h>
using namespace std;
int main() {
    string code;
    cin>>code;
    int i=0;
    string sum="";
    while(i<code.size()){
        if(code[i]=='-'){
            if(code[i+1]=='-'){
                sum.push_back('2');
            }else if(code[i+1]=='.'){
                sum.push_back('1');
            }
            i+=2;
        }else{
            sum.push_back('0');
            i++;
        }
    }
    cout<<sum<<endl;
    return 0;
}
