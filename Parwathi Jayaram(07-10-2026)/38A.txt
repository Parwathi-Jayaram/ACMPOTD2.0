So for this question I just kept adding the di values from a-1 to b-1.
#include <bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin>>n;
    vector<int> d;
    for(int i=0;i<n-1;i++){
        int a;
        cin>>a;
        d.push_back(a);
    }
    int a,b;
    cin>>a>>b;
    int op=0;
    for(int i=a-1;i<b-1;i++){
        op+=d[i];
    }
    cout<<op<<endl;
    return 0;
}
