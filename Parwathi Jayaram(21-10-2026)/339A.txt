So for this question I counted the number of 1s,2s and 3s. 
Then accordingly created the output string.
#include <bits/stdc++.h>
using namespace std;
int main() {
    map<int,int> vc;
    string s;
    cin>>s;
    for(int i=0;i<s.size();i++){
        vc[s[i]]++;
    }
    string op="";
    for(int i=0;i<vc['1'];i++){
        op.push_back('1');
        op.push_back('+');
    }
    for(int i=0;i<vc['2'];i++){
        op.push_back('2');
        op.push_back('+');
    }
    for(int i=0;i<vc['3'];i++){
        op.push_back('3');
        op.push_back('+');
    }
    op.pop_back();
    cout<<op<<endl;
    return 0;
}
