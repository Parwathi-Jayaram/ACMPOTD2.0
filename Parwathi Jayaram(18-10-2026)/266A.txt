So for this question I just counted number of adjacent stones iwth same colours.
#include <bits/stdc++.h>
using namespace std;
int main() {
    int nu;
    cin>>nu;
    if(nu==1){
        cout<<0<<endl;
        return 0;
    }
    int count=0;
    string s;
    cin>>s;
    for(int i=1;i<nu;i++){
        if(s[i]==s[i-1]){
            count++;
        }
    }
    cout<<count<<endl;
    return 0;
}
