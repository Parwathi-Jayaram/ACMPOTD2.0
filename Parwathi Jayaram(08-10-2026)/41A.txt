So for this question I actually first checked if the length of s and t are same.
Then I checked if t is the same as reversed s.
#include <bits/stdc++.h>
using namespace std;
int main() {
    string s,t;
    cin>>s;
    cin>>t;
    if(s.size()!=t.size()){
        cout<<"NO"<<endl;
        return 0;
    }
    int j=s.size()-1;
    string op="YES";
    for(char c:t){
        if(c!=s[j]){
            op="NO";
            break;
        }
        j--;
    }
    cout<<op<<endl;
    return 0;
}
