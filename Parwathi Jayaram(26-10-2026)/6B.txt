So for this question I created a set to count unique colors.
Then I iterated through the matrix and checked the 4 adjacent elements.
I then output the size of the set.
#include <bits/stdc++.h>
using namespace std;
int main() {
    int n,m;
    char p;
    cin>>n>>m>>p;
    vector<vector<char>>v;
    for(int i=0;i<n;i++){
        vector<char> a;
        for(int j=0;j<m;j++){
            char col;
            cin>>col;
            a.push_back(col);
        }
        v.push_back(a);
    }
    set<char> s;
    for(int i=0;i<n;i++){
        for(int j=0;j<m;j++){
            if(v[i][j]==p){
                if(j-1>=0 && v[i][j-1]!=v[i][j] && v[i][j-1]!='.'){
                    s.insert(v[i][j-1]);
                }
                if(j+1<m && v[i][j+1]!=v[i][j] && v[i][j+1]!='.'){
                    s.insert(v[i][j+1]);
                }
                if(i-1>=0 && v[i-1][j]!=v[i][j]&& v[i-1][j]!='.'){
                    s.insert(v[i-1][j]);
                }
                if(i+1<n && v[i+1][j]!=v[i][j] && v[i+1][j]!='.'){
                    s.insert(v[i+1][j]);
                }
            }
        }
    }
    cout<<s.size()<<endl;
    return 0;
}
