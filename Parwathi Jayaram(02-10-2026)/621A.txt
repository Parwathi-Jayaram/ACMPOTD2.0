So for this question while getting the input I actually calculated the minimum odd number and total sum simultanousl. 
If the sum was even I output it as it is other wise I subtracted the minimum odd from the sum and then output it.

#include <bits/stdc++.h>
using namespace std;
int main() {
    int n;
    cin>>n;
    vector<int> st;
    long long sum=0;
    long long mino=LLONG_MAX;
    for(int i=0;i<n;i++){
        long long a;
        cin>>a;
        if(a%2!=0){
            mino=min(mino,a);
        }
        sum+=a;
        st.push_back(a);
    }
    if(sum%2==0){
        cout<<sum<<endl;
    }else{
        cout<<sum-mino<<endl;
    }
    return 0;
}
